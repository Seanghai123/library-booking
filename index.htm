function doPost(e) { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName("BookingSheet"); // Check duplicate var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { var d = Utilities.formatDate( new Date(rows[i][0]),"GMT+7","dd/MM/yyyy"); if (d===data.date && rows[i][2]===data.time && rows[i][6]===data.room) { return ContentService .createTextOutput(JSON.stringify( {status:"duplicate"})) .setMimeType(ContentService.MimeType.JSON); } } sheet.appendRow([ data.date, data.day, data.time, data.topic, data.group, data.members, data.room, "កក់ហើយ", "" ]); return ContentService .createTextOutput(JSON.stringify({status:"ok"})) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName("BookingSheet"); var rows = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < rows.length; i++) { if (!rows[i][0]) continue; result.push({ date: rows[i][0], day: rows[i][1], time: rows[i][2], topic: rows[i][3], group: rows[i][4], members: rows[i][5], room: rows[i][6], status: rows[i][7] }); } return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
